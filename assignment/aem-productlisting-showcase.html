package com.adobe.aem.guides.wknd.core.dto;

import java.io.Serializable;
import java.util.Objects;

import com.google.gson.annotations.SerializedName;

/**
 * Data Transfer Object representing a Product.
 * Matches the required product listing JSON structure.
 */
public class Product implements Serializable {

    private static final long serialVersionUID = 1L;

    @SerializedName("id")
    private Long id;

    @SerializedName("name")
    private String name;

    @SerializedName("price")
    private Double price;

    @SerializedName("currency")
    private String currency;

    @SerializedName("image_url")
    private String imageUrl;

    public Product() {
    }

    public Product(Long id, String name, Double price, String currency, String imageUrl) {
        this.id = id;
        this.name = name;
        this.price = price;
        this.currency = currency;
        this.imageUrl = imageUrl;
    }

    public Long getId() {
        return id;
    }

    public void setId(Long id) {
        this.id = id;
    }

    public String getName() {
        return name;
    }

    public void setName(String name) {
        this.name = name;
    }

    public Double getPrice() {
        return price;
    }

    public void setPrice(Double price) {
        this.price = price;
    }

    public String getCurrency() {
        return currency;
    }

    public void setCurrency(String currency) {
        this.currency = currency;
    }

    public String getImageUrl() {
        return imageUrl;
    }

    public void setImageUrl(String imageUrl) {
        this.imageUrl = imageUrl;
    }

    public String getFormattedPrice() {
        if (price == null) {
            return "";
        }
        String curr = currency != null ? currency : "";
        return String.format("%.2f%s", price, curr);
    }

    @Override
    public boolean equals(Object o) {
        if (this == o) return true;
        if (o == null || getClass() != o.getClass()) return false;
        Product product = (Product) o;
        return Objects.equals(id, product.id) &&
               Objects.equals(name, product.name) &&
               Objects.equals(price, product.price) &&
               Objects.equals(currency, product.currency) &&
               Objects.equals(imageUrl, product.imageUrl);
    }

    @Override
    public int hashCode() {
        return Objects.hash(id, name, price, currency, imageUrl);
    }

    @Override
    public String toString() {
        return "Product{" +
                "id=" + id +
                ", name='" + name + '\'' +
                ", price=" + price +
                ", currency='" + currency + '\'' +
                ", imageUrl='" + imageUrl + '\'' +
                '}';
    }
}

import java.io.BufferedReader;

/**
 * Implementation of {@link ProductListingService}.
 * Fetches product catalog from an external REST API or falls back to
 * a resilient local seed dataset.
 */
@Component(
        service = ProductListingService.class,
        immediate = true
)
@Designate(ocd = ProductListingServiceConfig.class)
public class ProductListingServiceImpl implements ProductListingService {

    private static final Logger LOG = LoggerFactory.getLogger(ProductListingServiceImpl.class);

    private static final String DEFAULT_SEED_JSON = "["

            + "{\"id\":8,\"name\":\"Nut Mix\",\"price\":5.99,\"currency\":\"USD\",\"image_url\":\"http://dummyimage.com/117x239.png/dddddd/000000\"},"
            + "{\"id\":9,\"name\":\"Woven Storage Baskets\",\"price\":24.99,\"currency\":\"USD\",\"image_url\":\"http://dummyimage.com/235x223.png/dddddd/000000\"},"
            + "{\"id\":10,\"name\":\"Guitar Tuner\",\"price\":19.99,\"currency\":\"USD\",\"image_url\":\"http://dummyimage.com/232x130.png/5fa2dd/ffffff\"}"
            + "]";

    private final Gson gson = new GsonBuilder().create();
    private final Type productListType = new TypeToken<List<Product>>() {}.getType();

    private ProductListingServiceConfig config;

    @Activate
    @Modified
    public void activate(final ProductListingServiceConfig config) {
        this.config = config;
        LOG.info("ProductListingServiceImpl activated with endpoint: {}, default limit: {}",
                config != null ? config.apiEndpointUrl() : "none",
                config != null ? config.defaultLimit() : 9);
    }

    @Override
    public List<Product> getProducts() {
        int limit = (config != null && config.defaultLimit() > 0) ? config.defaultLimit() : 9;
        return getProducts(limit);
    }

    @Override
    public List<Product> getProducts(int limit) {
        List<Product> products = null;

        if (config != null && StringUtils.isNotBlank(config.apiEndpointUrl())) {
            products = fetchFromExternalEndpoint(config.apiEndpointUrl());
        }

        if (products == null || products.isEmpty()) {
            LOG.debug("Using local seed dataset for products");
            products = parseJson(DEFAULT_SEED_JSON);
        }

        if (limit > 0 && products.size() > limit) {
            return products.stream().limit(limit).collect(Collectors.toList());
        }

        return products;
    }

    @Override
    public String getProductsAsJson(int limit) {
        List<Product> products = getProducts(limit);
        return gson.toJson(products);
    }

    private List<Product> fetchFromExternalEndpoint(String endpointUrl) {
        HttpURLConnection connection = null;
        try {
            URL url = new URL(endpointUrl);
            connection = (HttpURLConnection) url.openConnection();
            connection.setRequestMethod("GET");
            connection.setRequestProperty("Accept", "application/json");

            int connectTimeout = (config != null && config.connectionTimeoutMs() > 0) ? config.connectionTimeoutMs() : 5000;
            int readTimeout = (config != null && config.readTimeoutMs() > 0) ? config.readTimeoutMs() : 5000;
            connection.setConnectTimeout(connectTimeout);
            connection.setReadTimeout(readTimeout);

            int responseCode = connection.getResponseCode();
            if (responseCode == HttpURLConnection.HTTP_OK) {
                try (BufferedReader reader = new BufferedReader(
                        new InputStreamReader(connection.getInputStream(), StandardCharsets.UTF_8))) {
                    StringBuilder response = new StringBuilder();
                    String line;
                    while ((line = reader.readLine()) != null) {
                        response.append(line);
                    }
                    return parseJson(response.toString());
                }
            } else {
                LOG.warn("Failed to fetch products from {}: HTTP status {}", endpointUrl, responseCode);
            }
        } catch (Exception e) {
            LOG.error("Exception while fetching products from external endpoint: {}", endpointUrl, e);
        } finally {
            if (connection != null) {
                connection.disconnect();
            }
        }
        return Collections.emptyList();
    }

    private List<Product> parseJson(String json) {
        try {
            List<Product> list = gson.fromJson(json, productListType);
            return list != null ? list : new ArrayList<>();
        } catch (Exception e) {
            LOG.error("Failed to parse product JSON", e);
            return Collections.emptyList();
        }
    }
}

import java.io.IOException;

import javax.servlet.Servlet;
import javax.servlet.ServletException;
import javax.servlet.http.HttpServletResponse;

import org.apache.commons.lang3.StringUtils;
import org.apache.sling.api.SlingHttpServletRequest;
import org.apache.sling.api.SlingHttpServletResponse;
import org.apache.sling.api.servlets.HttpConstants;
import org.apache.sling.api.servlets.SlingSafeMethodsServlet;
import org.apache.sling.servlets.annotations.SlingServletResourceTypes;
import org.osgi.service.component.annotations.Component;
import org.osgi.service.component.annotations.Reference;
import org.osgi.service.component.propertytypes.ServiceDescription;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import com.adobe.aem.guides.wknd.core.services.ProductListingService;

/**
 * Sling Servlet exposing the product catalog as a JSON endpoint.
 *
 * <p>Can be invoked either via:</p>
 * <ul>
 *   <li><b>Resource type selector:</b> <code>&lt;path-to-productlisting-component&gt;.products.json</code></li>
 *   <li><b>Path-based endpoint:</b> <code>/bin/wknd/products</code></li>
 * </ul>
 *
 * Supports query parameter: <code>limit</code> (default 9 for 3x3 grid).
 */
@Component(
        service = { Servlet.class },
        property = {
                "sling.servlet.paths=/bin/wknd/products",
                "sling.servlet.methods=" + HttpConstants.METHOD_GET
        }
)
@SlingServletResourceTypes(
        resourceTypes = "wknd/components/productlisting",
        methods = HttpConstants.METHOD_GET,
        selectors = "products",
        extensions = "json"
)
@ServiceDescription("WKND Product Listing JSON Servlet")
public class ProductListingServlet extends SlingSafeMethodsServlet {

    private static final long serialVersionUID = 1L;
    private static final Logger LOG = LoggerFactory.getLogger(ProductListingServlet.class);
    private static final int DEFAULT_LIMIT = 9;
    private static final int MAX_LIMIT = 100;

    @Reference
    private transient ProductListingService productListingService;

    @Override
    protected void doGet(final SlingHttpServletRequest request,
                         final SlingHttpServletResponse response) throws ServletException, IOException {
        response.setContentType("application/json;charset=UTF-8");
        response.setCharacterEncoding("UTF-8");
        response.setHeader("Cache-Control", "public, max-age=60");

        try {
            int limit = parseLimit(request.getParameter("limit"));
            LOG.debug("ProductListingServlet handling request with limit={}", limit);

            String jsonResponse = productListingService.getProductsAsJson(limit);

            response.setStatus(HttpServletResponse.SC_OK);
            response.getWriter().write(jsonResponse);
        } catch (NumberFormatException e) {
            LOG.warn("Invalid limit parameter provided", e);
            response.setStatus(HttpServletResponse.SC_BAD_REQUEST);
            response.getWriter().write("{\"error\":true,\"message\":\"Invalid 'limit' query parameter. Must be an integer.\"}");
        } catch (Exception e) {
            LOG.error("Error generating product listing response", e);
            response.setStatus(HttpServletResponse.SC_INTERNAL_SERVER_ERROR);
            response.getWriter().write("{\"error\":true,\"message\":\"Unable to load product catalog.\"}");
        }
    }

    private int parseLimit(String limitParam) {
        if (StringUtils.isBlank(limitParam)) {
            return DEFAULT_LIMIT;
        }
        int parsed = Integer.parseInt(limitParam.trim());
        if (parsed <= 0) {
            return DEFAULT_LIMIT;
        }
        return Math.min(parsed, MAX_LIMIT);
    }
}

package com.adobe.aem.guides.wknd.core.models;

import java.util.Collections;
import java.util.List;

import javax.annotation.PostConstruct;

import org.apache.commons.lang3.StringUtils;
import org.apache.sling.api.SlingHttpServletRequest;
import org.apache.sling.api.resource.Resource;
import org.apache.sling.models.annotations.Default;
import org.apache.sling.models.annotations.DefaultInjectionStrategy;
import org.apache.sling.models.annotations.Model;
import org.apache.sling.models.annotations.injectorspecific.OSGiService;
import org.apache.sling.models.annotations.injectorspecific.ValueMapValue;
import org.slf4j.Logger;
import org.slf4j.LoggerFactory;

import com.adobe.aem.guides.wknd.core.dto.Product;
import com.adobe.aem.guides.wknd.core.services.ProductListingService;

/**
 * Sling Model for the Product Listing Component.
 *
 * Exposes authored component configuration, grid dimensions, endpoint details,
 * and optional initial product data for progressive enhancement.
 */
@Model(
        adaptables = { Resource.class, SlingHttpServletRequest.class },
        defaultInjectionStrategy = DefaultInjectionStrategy.OPTIONAL,
        resourceType = "wknd/components/productlisting"
)
public class ProductListingModel {

    private static final Logger LOG = LoggerFactory.getLogger(ProductListingModel.class);
    private static final String DEFAULT_TITLE = "Product Listing Page using Html/CSS/JS";
    private static final String DEFAULT_ENDPOINT = "/bin/wknd/products";
    private static final int DEFAULT_LIMIT = 9;
    private static final int DEFAULT_COLUMNS = 3;

    @ValueMapValue
    @Default(values = DEFAULT_TITLE)
    private String title;

    @ValueMapValue
    @Default(values = "3x3")
    private String gridPreset;

    @ValueMapValue
    @Default(values = "osgi")
    private String sourceMode;

    @ValueMapValue
    @Default(values = DEFAULT_ENDPOINT)
    private String apiEndpoint;

    @ValueMapValue
    @Default(intValues = DEFAULT_LIMIT)
    private int limit;

    @ValueMapValue
    @Default(intValues = DEFAULT_COLUMNS)
    private int columns;

    @OSGiService
    private ProductListingService productListingService;

    private List<Product> initialProducts;

    @PostConstruct
    protected void init() {
        if (StringUtils.isBlank(title)) {
            title = DEFAULT_TITLE;
        }

        // Apply Grid Preset presets if configured
        if ("3x3".equalsIgnoreCase(gridPreset)) {
            this.columns = 3;
            this.limit = 9;
        } else if ("3x2".equalsIgnoreCase(gridPreset)) {
            this.columns = 3;
            this.limit = 6;
        } else if ("2x2".equalsIgnoreCase(gridPreset)) {
            this.columns = 2;
            this.limit = 4;
        } else if ("3x1".equalsIgnoreCase(gridPreset)) {
            this.columns = 3;
            this.limit = 3;
        } else {
            // "custom" or unspecified
            if (this.columns <= 0) {
                this.columns = DEFAULT_COLUMNS;
            }
            if (this.limit <= 0) {
                this.limit = DEFAULT_LIMIT;
            }
        }

        // Apply Data Source Strategy
        if ("osgi".equalsIgnoreCase(sourceMode) || StringUtils.isBlank(apiEndpoint)) {
            this.apiEndpoint = DEFAULT_ENDPOINT;
        }

        // Optional server-side prefetch via OSGi Service
        if (productListingService != null) {
            try {
                this.initialProducts = productListingService.getProducts(limit);
            } catch (Exception e) {
                LOG.error("Failed to load initial products in model init", e);
                this.initialProducts = Collections.emptyList();
            }
        } else {
            this.initialProducts = Collections.emptyList();
        }
    }

    public String getTitle() {
        return title;
    }

    public String getGridPreset() {
        return gridPreset;
    }

    public String getSourceMode() {
        return sourceMode;
    }

    public String getApiEndpoint() {
        return apiEndpoint;
    }

    public int getLimit() {
        return limit;
    }

    public int getColumns() {
        return columns;
    }

    public List<Product> getInitialProducts() {
        return initialProducts != null ? Collections.unmodifiableList(initialProducts) : Collections.emptyList();
    }

    public boolean hasProducts() {
        return initialProducts != null && !initialProducts.isEmpty();
    }

    public String getGridStyle() {
        return String.format("grid-template-columns: repeat(%d, 1fr);", columns > 0 ? columns : 3);
    }
}

/**
 * WKND Product Listing Component
 * Asynchronously fetches and renders product blocks into a 3x3 responsive grid.
 */
(function () {
    "use strict";

    class ProductListing {
        constructor(element) {
            this.container = element;
            this.endpoint = element.dataset.endpoint || "/bin/wknd/products";
            this.limit = parseInt(element.dataset.limit, 10) || 9;
            this.grid = element.querySelector(".cmp-productlisting__grid--active");
            this.loading = element.querySelector(".cmp-productlisting__loading");
            this.errorContainer = element.querySelector(".cmp-productlisting__status--error");
            this.errorMessage = element.querySelector(".cmp-productlisting__status-message");
            this.emptyContainer = element.querySelector(".cmp-productlisting__status--empty");
            this.retryBtn = element.querySelector(".cmp-productlisting__retry-btn");

            this.init();
        }

        init() {
            if (this.retryBtn) {
                this.retryBtn.addEventListener("click", () => this.fetchProducts());
            }

            this.fetchProducts();
        }

        /**
         * Fetch product data from the backend servlet/service.
         */
        async fetchProducts() {
            this.showLoading();

            const url = new URL(this.endpoint, window.location.origin);
            url.searchParams.set("limit", this.limit);

            const controller = new AbortController();
            const timeoutId = setTimeout(() => controller.abort(), 6000);

            try {
                const response = await fetch(url.toString(), {
                    method: "GET",
                    headers: {
                        "Accept": "application/json"
                    },
                    signal: controller.signal
                });

                clearTimeout(timeoutId);

                if (!response.ok) {
                    throw new Error(`HTTP error ${response.status}: ${response.statusText}`);
                }

                const data = await response.json();

                if (!Array.isArray(data)) {
                    throw new Error("Invalid response format: Expected JSON array of products.");
                }

                this.renderProducts(data);
            } catch (error) {
                clearTimeout(timeoutId);
                console.error("[ProductListing] Fetch error:", error);
                this.showError(error.name === "AbortError"
                    ? "Request timed out while loading products. Please try again."
                    : "Unable to load products. Please check the backend connection.");
            }
        }

        /**
         * Safely render product cards to prevent XSS.
         */
        renderProducts(products) {
            this.hideStatus();

            if (!this.grid) {
                return;
            }

            // Slice to exact limit (e.g. 9 for 3x3 blocks)
            const displayProducts = products.slice(0, this.limit);

            if (displayProducts.length === 0) {
                this.showEmpty();
                return;
            }

            // Clear previous cards
            this.grid.innerHTML = "";

            const fragment = document.createDocumentFragment();

            displayProducts.forEach((item) => {
                const card = document.createElement("article");
                card.className = "cmp-productlisting__card";
                card.setAttribute("tabindex", "0");
                card.setAttribute("role", "article");
                card.setAttribute("aria-label", item.name || "Product item");

                // Line 1: Product ID
                const idLine = document.createElement("p");
                idLine.className = "cmp-productlisting__card-line cmp-productlisting__card-id";
                idLine.textContent = `Product ID: ${item.id !== undefined ? item.id : ""}`;

                // Line 2: Product Name
                const nameLine = document.createElement("p");
                nameLine.className = "cmp-productlisting__card-line cmp-productlisting__card-name";
                nameLine.textContent = `Product Name :${item.name || ""}`;

                // Line 3: Price
                const priceLine = document.createElement("p");
                priceLine.className = "cmp-productlisting__card-line cmp-productlisting__card-price";
                const priceVal = typeof item.price === "number" ? item.price.toFixed(2) : (item.price || "");
                const currencyVal = item.currency || "";
                priceLine.textContent = `Price : ${priceVal}${currencyVal}`;

                card.appendChild(idLine);
                card.appendChild(nameLine);
                card.appendChild(priceLine);

                fragment.appendChild(card);
            });

            this.grid.appendChild(fragment);
            this.grid.style.display = "grid";
        }

        showLoading() {
            if (this.loading) this.loading.style.display = "block";
            if (this.grid) this.grid.style.display = "none";
            if (this.errorContainer) this.errorContainer.style.display = "none";
            if (this.emptyContainer) this.emptyContainer.style.display = "none";
        }

        hideStatus() {
            if (this.loading) this.loading.style.display = "none";
            if (this.errorContainer) this.errorContainer.style.display = "none";
            if (this.emptyContainer) this.emptyContainer.style.display = "none";
        }

        showError(message) {
            if (this.loading) this.loading.style.display = "none";
            if (this.grid) this.grid.style.display = "none";
            if (this.emptyContainer) this.emptyContainer.style.display = "none";

            if (this.errorContainer) {
                if (this.errorMessage) {
                    this.errorMessage.textContent = message;
                }
                this.errorContainer.style.display = "block";
            }
        }

        showEmpty() {
            if (this.loading) this.loading.style.display = "none";
            if (this.grid) this.grid.style.display = "none";
            if (this.errorContainer) this.errorContainer.style.display = "none";
            if (this.emptyContainer) this.emptyContainer.style.display = "block";
        }
    }

    /**
     * Auto-initialize all component instances on DOM ready.
     */
    function initAll() {
        const elements = document.querySelectorAll('[data-cmp-is="productlisting"]');
        elements.forEach((el) => {
            if (!el.__productListingInitialized) {
                el.__productListingInitialized = new ProductListing(el);
            }
        });
    }

    if (document.readyState === "loading") {
        document.addEventListener("DOMContentLoaded", initAll);
    } else {
        initAll();
    }
})();

Architecture & Best Practices Dossier

Layer / Question Enterprise Solution Design Rationale
Hardcoded vs. URL Configurable OSGi Service + Resilient Seed Fallback Hardcoding in servlets is an anti-pattern. Relying solely on external endpoints risks downtime and CORS failures. Our OSGi Service attempts external REST calls with timeouts, gracefully falling back to seed data.
Sling Servlet Dual Registration: ResourceType + /bin/wknd/products ResourceType servlets provide security and dispatcher cacheability, while direct path mounts support standalone AJAX calls.
Sling Model ProductListingModel with JCR Injection Injects authored title, grid presets (3x3, 2x2, etc.), and endpoint from Coral UI 3 dialog, computing dynamic CSS styles.
Security (XSS) document.createElement & Context Escaping Zero raw innerHTML string concatenation; immune to malicious script injection in product names or prices.
Unit Testing JUnit 5 + AEM Mocks (17 tests passing) Mocks OSGi service injection, request lifecycle, bounds checking, and Sling Model adaptation.
